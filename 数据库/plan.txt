三、数据库建表 SQL
1. 用户表 User
sql
CREATE TABLE User(
Id INT PRIMARY KEY AUTO_INCREMENT,
UserName VARCHAR(50) NOT NULL UNIQUE COMMENT '用户名/学工号',
Password VARCHAR(100) NOT NULL COMMENT '密码',
RealName VARCHAR(50) NOT NULL COMMENT '真实姓名',
Role TINYINT NOT NULL DEFAULT 1 COMMENT '1普通用户2普通管理员3超级管理员',
Email VARCHAR(100) NULL COMMENT '绑定邮箱',
IsActive TINYINT NOT NULL DEFAULT 1 COMMENT '1正常0禁用'
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;
2. 活动申请表 ActivityApply
sql
CREATE TABLE ActivityApply(
Id INT PRIMARY KEY AUTO_INCREMENT,
Title VARCHAR(100) NOT NULL COMMENT '活动名称',
Content TEXT NOT NULL COMMENT '活动详情',
PublisherUserName VARCHAR(50) NOT NULL COMMENT '发布者用户名',
AuditStatus TINYINT NOT NULL DEFAULT 0 COMMENT '0待审核1通过2驳回',
RejectReason TEXT NULL COMMENT '驳回理由',
CreateTime DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP COMMENT '申请时间',
MaxCount INT NOT NULL COMMENT '最大参与人数',
CurrentCount INT NOT NULL DEFAULT 0 COMMENT '当前报名人数',
SignInCount INT NOT NULL DEFAULT 0 COMMENT '已签到人数',
ActivityStartTime DATETIME NOT NULL COMMENT '活动开始时间',
ActivityEndTime DATETIME NOT NULL COMMENT '活动结束时间',
EnrollStartTime DATETIME NOT NULL COMMENT '报名开始时间',
EnrollEndTime DATETIME NOT NULL COMMENT '报名结束时间'
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;
3. 正式活动表 Activity
sql
CREATE TABLE Activity(
Id INT PRIMARY KEY AUTO_INCREMENT,
Title VARCHAR(100) NOT NULL COMMENT '活动名称',
Content TEXT NOT NULL COMMENT '活动详情',
PublisherUserName VARCHAR(50) NOT NULL COMMENT '发布者用户名',
AuditStatus TINYINT NOT NULL DEFAULT 1 COMMENT '1通过',
RejectReason TEXT NULL COMMENT '驳回理由',
CreateTime DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间',
MaxCount INT NOT NULL COMMENT '最大参与人数',
CurrentCount INT NOT NULL DEFAULT 0 COMMENT '当前报名人数',
SignInCount INT NOT NULL DEFAULT 0 COMMENT '已签到人数',
ActivityStartTime DATETIME NOT NULL COMMENT '活动开始时间',
ActivityEndTime DATETIME NOT NULL COMMENT '活动结束时间',
EnrollStartTime DATETIME NOT NULL COMMENT '报名开始时间',
EnrollEndTime DATETIME NOT NULL COMMENT '报名结束时间'
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;
4. 留言表 Message
sql
CREATE TABLE Message(
Id INT PRIMARY KEY AUTO_INCREMENT,
ActivityId INT NOT NULL COMMENT '正式活动ID',
UserName VARCHAR(50) NOT NULL COMMENT '留言用户',
Content TEXT NOT NULL COMMENT '留言内容',
CreateTime DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP COMMENT '留言时间'
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;
5. 报名表 Enroll
sql
CREATE TABLE Enroll(
Id INT PRIMARY KEY AUTO_INCREMENT,
UserName VARCHAR(50) NOT NULL COMMENT '报名用户',
ActivityId INT NOT NULL COMMENT '正式活动ID',
EnrollTime DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP COMMENT '报名时间'
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;
6. 签到表 SignIn
sql
CREATE TABLE SignIn(
Id INT PRIMARY KEY AUTO_INCREMENT,
UserName VARCHAR(50) NOT NULL COMMENT '签到用户',
ActivityId INT NOT NULL COMMENT '正式活动ID',
SignInTime DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP COMMENT '签到时间'
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;
7. 签到码表 ActivitySignInCode
sql
CREATE TABLE ActivitySignInCode(
Id INT PRIMARY KEY AUTO_INCREMENT,
Code VARCHAR(20) NOT NULL UNIQUE COMMENT '签到码',
ActivityId INT NOT NULL COMMENT '正式活动ID',
ExpireTime DATETIME NOT NULL COMMENT '过期时间'
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;
8. 密码重置申请表 ResetPwdApply
sql
CREATE TABLE ResetPwdApply(
Id INT PRIMARY KEY AUTO_INCREMENT,
UserName VARCHAR(50) NOT NULL COMMENT '申请用户',
RealName VARCHAR(50) NOT NULL COMMENT '真实姓名',
Reason TEXT NOT NULL COMMENT '申请理由',
Status TINYINT NOT NULL DEFAULT 0 COMMENT '0待审核1通过2拒绝',
RejectReason TEXT NULL COMMENT '拒绝理由'
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;
